// ── Enums ──────────────────────────────────────────────────

Enum platform_role {
  ADMIN     // співробітник Sentio
  USER      // звичайний користувач
}

Enum org_role {
  OWNER     // власник бюро, керує підпискою і учасниками
  LAWYER    // адвокат: повний доступ до своїх справ
  ASSISTANT // помічник: без доступу до гонорарів
}

Enum auth_provider {
  LOCAL
  GOOGLE
  APPLE
}

Enum client_type {
  INDIVIDUAL    // фізична особа
  SOLE_TRADER   // ФОП
  COMPANY       // юридична особа
}

Enum procedure_type {
  CIVIL          // ЦПК
  ADMINISTRATIVE // КАС
  COMMERCIAL     // ГПК
  CRIMINAL       // КПК
  OTHER
}

Enum case_status {
  DRAFT
  PRE_TRIAL      // досудова стадія
  FIRST_INSTANCE
  APPEAL
  CASSATION
  ENFORCEMENT    // виконавче провадження
  CLOSED
  ARCHIVED
}

Enum day_kind {
  CALENDAR
  WORKING
}

Enum deadline_status {
  PENDING
  DONE
  MISSED
  SUSPENDED   // строк зупинено
  EXTENDED    // строк поновлено ухвалою
}

Enum notification_channel {
  EMAIL
  TELEGRAM
  PUSH
  IN_APP
}

Enum notification_status {
  QUEUED
  SENT
  FAILED
  READ
}

Enum plan_tier {
  SOLO
  BUREAU
  FIRM
}

Enum subscription_status {
  TRIALING
  ACTIVE
  PAST_DUE
  CANCELED
}


// ── Організації та користувачі ─────────────────────────────

Table organizations {
  id bigserial [pk]
  name varchar(255) [not null]
  slug varchar(100) [not null, unique]
  edrpou varchar(10)              // ЄДРПОУ бюро, якщо юрособа
  plan plan_tier [not null, default: `SOLO`]
  subscription_status subscription_status [not null, default: `TRIALING`]
  trial_ends_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
  deleted_at timestamptz
}

Table users {
  id bigserial [pk]
  email citext [not null, unique]      // citext, щоб не ловити регістр вручну
  password_hash varchar(72)            // null для чистого OAuth; bcrypt = 60 символів
  phone_number varchar(20) [unique]    // E.164, nullable — OAuth номер не дає
  platform_role platform_role [not null, default: `USER`]

  last_name varchar(100)
  first_name varchar(100)
  middle_name varchar(100)

  email_verified_at timestamptz
  phone_verified_at timestamptz
  last_login_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
  deleted_at timestamptz
}

// Один акаунт може мати кілька способів входу (local + Google)
Table user_identities {
  id bigserial [pk]
  user_id bigint [not null, ref: > users.id]
  provider auth_provider [not null]
  provider_user_id varchar(255) [not null]  // sub із токена
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (provider, provider_user_id) [unique]
    (user_id, provider) [unique]
  }
}

Table organization_members {
  id bigserial [pk]
  organization_id bigint [not null, ref: > organizations.id]
  user_id bigint [not null, ref: > users.id]
  role org_role [not null, default: `LAWYER`]
  invited_by bigint [ref: > users.id]
  joined_at timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes {
    (organization_id, user_id) [unique]
    organization_id
    user_id
  }
}

Table refresh_tokens {
  id bigserial [pk]
  user_id bigint [not null, ref: > users.id]
  token_hash varchar(64) [not null, unique]  // зберігаємо хеш, не токен
  user_agent varchar(255)
  ip inet
  expires_at timestamptz [not null]
  revoked_at timestamptz
  created_at timestamptz [not null, default: `now()`]

  indexes { user_id }
}


// ── Клієнти ────────────────────────────────────────────────

Table clients {
  id bigserial [pk]
  organization_id bigint [not null, ref: > organizations.id]
  type client_type [not null, default: `INDIVIDUAL`]

  // фізична особа / ФОП
  last_name varchar(100)
  first_name varchar(100)
  middle_name varchar(100)
  birth_date date
  rnokpp varchar(10)              // РНОКПП (ІПН)
  passport varchar(50)

  // юридична особа
  company_name varchar(255)
  edrpou varchar(10)
  director_name varchar(255)      // для преамбули договорів

  email varchar(255)
  phone_number varchar(20)
  address text
  notes text

  created_by bigint [not null, ref: > users.id]
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
  deleted_at timestamptz

  indexes {
    (organization_id, last_name)
    (organization_id, rnokpp)
    (organization_id, edrpou)
    organization_id
  }

  Note: '''
  CHECK: type = COMPANY  => company_name IS NOT NULL
         type <> COMPANY => last_name IS NOT NULL AND first_name IS NOT NULL
  Пошук за прізвищем — окремий GIN-індекс на pg_trgm, бо потрібен нечіткий збіг.
  '''
}


// ── Суди і справи ──────────────────────────────────────────

Table courts {
  id bigserial [pk]
  name varchar(255) [not null]
  code varchar(20) [unique]      // код суду з номера справи: 761, 522
  instance smallint [not null]   // 1 / 2 / 3
  region varchar(100)
  address text
  is_active boolean [not null, default: true]
}

Table cases {
  id bigserial [pk]
  organization_id bigint [not null, ref: > organizations.id]
  client_id bigint [not null, ref: > clients.id]
  responsible_user_id bigint [not null, ref: > users.id]

  case_number varchar(50)        // 761/4823/25 — null поки не відкрито провадження
  internal_number varchar(50)    // власна нумерація бюро
  title varchar(255) [not null]
  procedure procedure_type [not null]
  status case_status [not null, default: `DRAFT`]
  court_id bigint [ref: > courts.id]
  judge_name varchar(255)
  client_role varchar(50)        // позивач / відповідач / третя особа
  opened_at date
  closed_at date

  registry_watch_enabled boolean [not null, default: false]
  registry_last_checked_at timestamptz

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]
  deleted_at timestamptz

  indexes {
    (organization_id, status)
    (organization_id, case_number)
    client_id
    responsible_user_id
    (registry_watch_enabled, registry_last_checked_at)  // черга поллінгу
  }
}

// Хронологія справи — фіксує події, що запускають відлік строків
Table case_events {
  id bigserial [pk]
  case_id bigint [not null, ref: > cases.id]
  event_code varchar(50) [not null]   // 'CLAIM_FILED', 'RULING_RECEIVED'
  title varchar(255) [not null]
  description text
  occurred_on date [not null]         // дата події
  received_on date                    // дата отримання — від неї часто йде строк
  source varchar(20) [not null, default: 'MANUAL']  // MANUAL | REGISTRY
  registry_document_id bigint [ref: > registry_documents.id]
  created_by bigint [ref: > users.id]
  created_at timestamptz [not null, default: `now()`]

  indexes { (case_id, occurred_on) }
}

Table hearings {
  id bigserial [pk]
  case_id bigint [not null, ref: > cases.id]
  scheduled_at timestamptz [not null]
  kind varchar(50)                 // підготовче / по суті / апеляційне
  courtroom varchar(50)
  outcome text                     // заповнюється після засідання -> йде у звіт клієнту
  reported_to_client_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes { (case_id, scheduled_at) }
}


// ── Строки: ядро продукту ──────────────────────────────────

// Довідник правил. Версіонований — редакції кодексів змінюються,
// і старі справи не мають "переїжджати" на нові строки.
Table deadline_rules {
  id bigserial [pk]
  code varchar(50) [not null]        // 'CPC_STATEMENT_OF_DEFENCE'
  procedure procedure_type [not null]
  trigger_event_code varchar(50) [not null]  // яка подія запускає відлік
  title varchar(255) [not null]
  legal_basis varchar(255) [not null]        // 'ст. 178 ЦПК України'
  duration_value smallint [not null]
  duration_unit varchar(10) [not null, default: 'DAY']  // DAY | MONTH
  day_kind day_kind [not null, default: `CALENDAR`]
  count_from varchar(20) [not null, default: 'NEXT_DAY'] // строк тече з наступного дня
  is_extendable boolean [not null, default: false]
  valid_from date [not null]
  valid_to date                       // null = чинне

  indexes { (code, valid_from) [unique] }
}

Table holidays {
  date date [pk]
  name varchar(100) [not null]
  is_working boolean [not null, default: false]  // перенесені робочі суботи
}

Table deadlines {
  id bigserial [pk]
  organization_id bigint [not null, ref: > organizations.id]
  case_id bigint [not null, ref: > cases.id]
  rule_id bigint [ref: > deadline_rules.id]     // null = ручний строк
  triggering_event_id bigint [ref: > case_events.id]

  title varchar(255) [not null]
  legal_basis varchar(255)          // знімок на момент розрахунку
  starts_on date [not null]
  due_on date [not null]            // порахована дата
  status deadline_status [not null, default: `PENDING`]
  completed_at timestamptz
  completed_by bigint [ref: > users.id]
  extended_to date
  note text

  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (organization_id, status, due_on)   // головний запит календаря
    (case_id, due_on)
    (status, due_on)                    // черга нагадувань
  }

  Note: 'legal_basis і due_on — денормалізований знімок правила. Зміна кодексу не переписує історію.'
}


// ── Моніторинг реєстру (ЄДРСР) ─────────────────────────────

Table registry_documents {
  id bigserial [pk]
  case_number varchar(50) [not null]
  registry_doc_id varchar(64) [not null, unique]  // id документа в ЄДРСР
  doc_type varchar(100)              // ухвала / рішення / постанова
  court_name varchar(255)
  adjudication_date date
  publication_date date
  url text
  raw_text text
  fetched_at timestamptz [not null, default: `now()`]

  indexes { case_number }
}


// ── Документи і шаблони ────────────────────────────────────

Table document_templates {
  id bigserial [pk]
  organization_id bigint [ref: > organizations.id]  // null = системний шаблон
  name varchar(255) [not null]
  procedure procedure_type
  category varchar(50)              // позов / скарга / клопотання / договір
  body text [not null]              // розмітка з плейсхолдерами
  variables jsonb [not null, default: '[]']  // опис полів для форми
  legal_basis varchar(255)
  is_active boolean [not null, default: true]
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes { (organization_id, category) }
}

Table documents {
  id bigserial [pk]
  organization_id bigint [not null, ref: > organizations.id]
  case_id bigint [ref: > cases.id]
  template_id bigint [ref: > document_templates.id]
  name varchar(255) [not null]
  storage_key varchar(512) [not null]   // ключ в S3, не сам файл
  mime_type varchar(100)
  size_bytes bigint
  filled_values jsonb                   // чим заповнили шаблон — для перегенерації
  version smallint [not null, default: 1]
  uploaded_by bigint [not null, ref: > users.id]
  created_at timestamptz [not null, default: `now()`]
  deleted_at timestamptz

  // indexes { case_id, organization_id }
}


// ── Сповіщення ─────────────────────────────────────────────

Table notification_settings {
  id bigserial [pk]
  user_id bigint [not null, ref: > users.id]
  channel notification_channel [not null]
  destination varchar(255)          // telegram chat_id, email
  is_enabled boolean [not null, default: true]
  lead_days smallint[] [default: `'{14,7,3,1}'`]  // за скільки днів нагадувати
  quiet_hours_start time
  quiet_hours_end time

  indexes { (user_id, channel) [unique] }
}

Table notifications {
  id bigserial [pk]
  user_id bigint [not null, ref: > users.id]
  deadline_id bigint [ref: > deadlines.id]
  case_id bigint [ref: > cases.id]
  channel notification_channel [not null]
  status notification_status [not null, default: `QUEUED`]
  subject varchar(255)
  body text
  scheduled_for timestamptz [not null]
  sent_at timestamptz
  error text
  dedup_key varchar(128) [unique]   // захист від дублів при ретраях
  created_at timestamptz [not null, default: `now()`]

  // indexes { (status, scheduled_for), user_id }
}

}
